```python
# 1. Imports section

# 2. Functions section
def insere(liste, elem):
    """
    Fonction qui inset un élement à sa bonne place dans une liste.
    """
    # First of all, we search the position for insering the element.
    i = 0
    while i < len(liste) and liste[i] < elem :
        i +=1
    # We insert elem in i position if it is not the last of the list
    if i < len(liste) :
        liste.insert(i,elem)
    # We insert elem at the end if it is the bigger element into the list
    else:
        liste.append(elem)

    return liste


def tri_par_insertion(liste):
    """
    Fonction qui trie une liste par insertion
    """
    liste_triee = []
    for el in liste :
        liste_triee = insere(liste_triee, el)
    return liste_triee

# 3. Main routine section
if __name__ == "__main__":
    print(insere([1,2,4,5,6,9,10], 11))
    print(tri_par_insertion([10,9,8,7,3,12,13,5,4,1,11,14,2]))
```

```python
def alterne(liste_1,liste_2):
    """
    Fonction qui renvoie une liste contenant en alternance les éléments de la 
    première liste et ceux de la deuxième liste
    """
    # first check : the list need to be the same size
    if len(liste_1) == len(liste_2):
        liste_res = []
        i=0
        while i<len(liste_1):
            # alternatively add one element from each list
            liste_res.append(liste_1[i])
            liste_res.append(liste_2[i])
            i += 1
        return liste_res
    else:
        return []

# 3. Main routine section
```

```python
def compress_adn (str_adn) :
    """
    compresse une str_adn 'GAAATTGC' -> ['G', 'A', 3, 'T', 2, 'G', 'C']
    """ 
    res = []
    i = 0
    while i < len(str_adn):
        nb = str_adn[i]
        compteur = 1
        j = i + 1
        while j < len(str_adn) and str_adn[j] == str_adn[i]:  # same string repeats
            compteur += 1
            j += 1
        res += [nb]
        if compteur > 1: # add the number to the compressed list
            res += [compteur]
        i = j
    return res

# 3. Main routine section
if __name__ == "__main__":
    chaine = 'GAAATTGC'
    print(compress_adn(chaine))
```
